    <script>
        // Tab switching
        const tabs = {
            text: document.getElementById('tab-text'),
            image: document.getElementById('tab-image'),
            doc: document.getElementById('tab-doc')
        };
        
        const panels = {
            text: document.getElementById('panel-text'),
            image: document.getElementById('panel-image'),
            doc: document.getElementById('panel-doc')
        };
        
        function switchTab(activeTab) {
            Object.keys(tabs).forEach(key => {
                if (key === activeTab) {
                    tabs[key].classList.remove('glass', 'text-gray-300');
                    tabs[key].classList.add('bg-gradient-to-r', 'from-primary', 'to-secondary', 'text-dark');
                    panels[key].classList.remove('hidden');
                } else {
                    tabs[key].classList.add('glass', 'text-gray-300');
                    tabs[key].classList.remove('bg-gradient-to-r', 'from-primary', 'to-secondary', 'text-dark');
                    panels[key].classList.add('hidden');
                }
            });
        }
        
        tabs.text.addEventListener('click', () => switchTab('text'));
        tabs.image.addEventListener('click', () => switchTab('image'));
        tabs.doc.addEventListener('click', () => switchTab('doc'));
        
        // Character count
        const textInput = document.getElementById('text-input');
        const charCount = document.getElementById('char-count');
        
        textInput.addEventListener('input', () => {
            charCount.textContent = textInput.value.length;
        });
        
        // Detection simulation
        const detectBtn = document.getElementById('detect-btn');
        const resultModal = document.getElementById('result-modal');
        const closeModal = document.getElementById('close-modal');
        const resultIcon = document.getElementById('result-icon');
        const resultText = document.getElementById('result-text');
        const resultDetails = document.getElementById('result-details');
        const aiProbability = document.getElementById('ai-probability');
        const probabilityBar = document.getElementById('probability-bar');
        const confidence = document.getElementById('confidence');
        
        detectBtn.addEventListener('click', () => {
            resultModal.classList.remove('hidden');
            resultDetails.classList.add('hidden');
            resultIcon.textContent = '⏳';
            resultText.textContent = '正在分析中，请稍候...';
            
            setTimeout(() => {
                const probability = Math.floor(Math.random() * 100);
                resultDetails.classList.remove('hidden');
                
                if (probability > 70) {
                    resultIcon.textContent = '🤖';
                    resultText.textContent = '高度疑似AI生成内容';
                } else if (probability > 30) {
                    resultIcon.textContent = '🤔';
                    resultText.textContent = '可能包含AI生成内容';
                } else {
                    resultIcon.textContent = '✅';
                    resultText.textContent = '极可能是人类创作内容';
                }
                
                aiProbability.textContent = probability + '%';
                probabilityBar.style.width = probability + '%';
                confidence.textContent = (90 + Math.floor(Math.random() * 9)) + '%';
            }, 2000);
        });
        
        closeModal.addEventListener('click', () => {
            resultModal.classList.add('hidden');
        });
        
        resultModal.addEventListener('click', (e) => {
            if (e.target === resultModal) {
                resultModal.classList.add('hidden');
            }
        });
        
        // File upload handling
        const dropZone = document.getElementById('drop-zone');
        const imageInput = document.getElementById('image-input');
        
        dropZone.addEventListener('click', () => imageInput.click());
        dropZone.addEventListener('dragover', (e) => {
            e.preventDefault();
            dropZone.classList.add('border-primary');
        });
        dropZone.addEventListener('dragleave', () => {
            dropZone.classList.remove('border-primary');
        });
        dropZone.addEventListener('drop', (e) => {
            e.preventDefault();
            dropZone.classList.remove('border-primary');
            // Handle file
        });
        
        imageInput.addEventListener('change', () => {
            // Handle file
        });
        
        // PWA registration
        if ('serviceWorker' in navigator) {
            navigator.serviceWorker.register('sw.js')
                .then(reg => console.log('Service Worker registered'))
                .catch(err => console.log('Service Worker registration failed'));
        }
    </script>
